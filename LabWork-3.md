```csharp
using System;

class Program
{
    static void Swap(int[] arr, int i, int j)
    {
        int temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }

    static void Sink(int[] arr, int i, int n)
    {
        int k = i;
        while (true)
        {
            int j = 2 * k + 1;
            if (j >= n)
                break;
            if (j + 1 < n && arr[j + 1] > arr[j])
                j++;
            if (arr[k] >= arr[j])
                break;
            Swap(arr, k, j);
            k = j;
        }
    }

    static int[] HeapSort(int[] arr)
    {
        int n = arr.Length;
        Console.WriteLine($"Початковий масив: [{string.Join(", ", arr)}]\n");
        Console.WriteLine("--- Фаза 1: Побудова максимальної купи ---");
        for (int i = n / 2 - 1; i >= 0; i--)
        {
            Console.WriteLine($"Занурюємо елемент з індексу {i}: {arr[i]}");
            Sink(arr, i, n);
        }
        Console.WriteLine($"\nМасив після побудови купи: [{string.Join(", ", arr)}]\n");
        Console.WriteLine("--- Фаза 2: Сортування ---");

        for (int i = n - 1; i > 0; i--)
        {
            Console.WriteLine($"Міняємо місцями корінь ({arr[0]}) та останній елемент ({arr[i]})");
            Swap(arr, 0, i);
            n--;
            Console.WriteLine($"Розмір купи зменшився до {n}. Відновлюємо властивості купи.");
            Sink(arr, 0, n);
            Console.WriteLine($"Масив на поточному кроці: [{string.Join(", ", arr)}]\n");
        }

        return arr;
    }

    static void Main()
    {
        int[] A = { 80, 27, 37, 36, 91, 53, 86, 66, 98 };
        int[] sortedA = HeapSort(A);
        Console.WriteLine($"Відсортований масив: [{string.Join(", ", sortedA)}]");
    }
}
```
